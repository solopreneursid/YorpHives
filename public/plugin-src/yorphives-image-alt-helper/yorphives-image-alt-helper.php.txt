if (!defined('ABSPATH')) { exit; }

add_action('add_meta_boxes', function () {
    foreach (array('post', 'page') as $type) {
        add_meta_box('yhiah-box', 'YorpHives Image Alt Helper', 'yhiah_render', $type, 'side', 'default');
    }
});

function yhiah_render($post) {
    if (!current_user_can('edit_post', $post->ID)) { return; }
    ?>
    <p>Review image alternative text in this draft. This tool does not generate descriptions or change your content.</p>
    <button type="button" class="button" id="yhiah-run">Review images</button>
    <div id="yhiah-results" aria-live="polite"><p>Run a review to see image checks.</p></div>
    <script>
    (function () {
        const button = document.getElementById('yhiah-run');
        if (!button) return;
        button.addEventListener('click', function () {
            const results = document.getElementById('yhiah-results');
            results.replaceChildren();
            try {
                const blockEditor = window.wp && window.wp.data && typeof window.wp.data.select === 'function' ? window.wp.data.select('core/editor') : null;
                const blockContent = blockEditor && typeof blockEditor.getEditedPostContent === 'function' ? String(blockEditor.getEditedPostContent() || '') : '';
                const editor = window.tinymce && window.tinymce.get('content');
                const textarea = document.getElementById('content');
                const html = blockContent || (editor && !editor.isHidden() ? editor.getContent() : (textarea ? textarea.value : ''));
                const doc = new DOMParser().parseFromString(html, 'text/html');
                const images = Array.from(doc.querySelectorAll('img'));
                const summary = document.createElement('p');
                summary.textContent = images.length ? 'Found ' + images.length + ' image(s).' : 'No images found in the current editor content.';
                results.append(summary);
                if (!images.length) return;
                const list = document.createElement('ul');
                images.forEach(function (img, index) {
                    const alt = img.getAttribute('alt');
                    const label = img.getAttribute('src') || 'Image ' + (index + 1);
                    const item = document.createElement('li');
                    const state = alt === null ? 'Missing alt attribute' : (alt.trim() === '' ? 'Empty alt text' : 'Alt text present');
                    item.textContent = state + ' — ' + label;
                    list.append(item);
                });
                results.append(list);
                const note = document.createElement('p');
                note.textContent = 'Decorative images may correctly use an empty alt attribute. Check each image’s purpose.';
                results.append(note);
            } catch (error) {
                results.textContent = 'Could not inspect editor content. Try switching to the Code editor or saving the draft first.';
            }
        });
    })();
    </script>
    <?php
}
